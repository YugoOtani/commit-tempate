async startRequest(
    mappingPath: string | undefined,
    executionMode: ExecutionMode,
    stopOnEntry: boolean
): Promise<string | EmdbResponseStartParam> {
    const res = await this.call('start', { mappingPath, executionMode, stopOnEntry }, 100000);
    if (res.kind !== 'start') {
        if (res.kind === 'commandFail') {
            const msg = res.params.message;
            return msg;
        }
        engineErr(`start request failed: ${JSON.stringify(res)}`);
    }
    try {
        return {
            nodes: res.params.nodes,
            edges: res.params.edges,
            inputNodes: res.params.inputNodes,
            outputNodes: res.params.outputNodes
        };
    } catch {
        engineErr(`start response parse error: ${JSON.stringify(res)}`);
        return "Internal error in start response";
    }
}
